// --- ИНИЦИАЛИЗАЦИЯ И ПЕРЕМЕННЫЕ --- let channels = JSON.parse(localStorage.getItem("channels")) || []; let customChannels = JSON.parse(localStorage.getItem("customChannels")) || []; let favorites = JSON.parse(localStorage.getItem("favorites")) || []; let selectedCountry = "all"; let hls = null; // Элементы плеера const video = document.getElementById("video") || document.querySelector("video"); const player = video; // Функция вывода статуса function status(msg) { const statusEl = document.getElementById("status"); if (statusEl) statusEl.textContent = msg; console.log(msg); } function showStatus(msg) { status(msg); } // Получение всех каналов function getAllChannels() { return [...channels, ...customChannels]; } function getChannelId(channel) { return channel.url || channel.name; } // Сохранение данных function saveData() { localStorage.setItem("channels", JSON.stringify(channels)); localStorage.setItem("customChannels", JSON.stringify(customChannels)); localStorage.setItem("favorites", JSON.stringify(favorites)); } // --- ФУНКЦИИ ВЫБОРА СТРАНЫ И РЕНДЕРИНГА --- function chooseCountry(country) { selectedCountry = country; document.querySelectorAll(".country").forEach(item => { item.classList.toggle("active", item.dataset.country === country); }); renderChannels(); } // Навешиваем клик на все кнопки стран document.querySelectorAll(".country").forEach(item => { item.addEventListener("click", () => { chooseCountry(item.dataset.country); }); }); // Рендеринг основного списка каналов function renderChannels() { const channelList = document.getElementById("channels"); if (!channelList) return; channelList.innerHTML = ""; const filtered = getAllChannels().filter(c => { if (!selectedCountry || selectedCountry === "all") return true; return c.country === selectedCountry; }); filtered.forEach((c) => { const card = document.createElement("div"); card.className = "channel-card"; const title = document.createElement("b"); title.textContent = c.name; const description = document.createElement("p"); description.textContent = c.description || "Телеканал EuroVoice"; const actions = document.createElement("div"); actions.className = "actions"; const playBtn = document.createElement("button"); playBtn.className = "btn"; playBtn.textContent = "Смотреть ▶"; playBtn.onclick = () => playStream(c.url); actions.appendChild(playBtn); card.append(title, description, actions); channelList.appendChild(card); }); renderFavorites(); } // Рендеринг блока избранных каналов function renderFavorites() { const box = document.getElementById("favoritesInfo"); if (!box) return; const favoriteChannels = getAllChannels().filter( channel => favorites.includes(getChannelId(channel)) ); box.innerHTML = ""; favoriteChannels.forEach(channel => { const item = document.createElement("div"); item.className = "channel-card"; const name = document.createElement("b"); name.textContent = channel.name; const button = document.createElement("button"); button.className = "btn"; button.textContent = "Смотреть ▶"; button.onclick = () => playStream(channel.url); item.append(name, button); box.appendChild(item); }); } // --- ДОБАВЛЕНИЕ И УДАЛЕНИЕ КАНАЛОВ --- function addChannel() { const nameEl = document.getElementById("newName"); const countryEl = document.getElementById("newCountry"); const urlEl = document.getElementById("newUrl"); if (!nameEl || !urlEl) return; const name = nameEl.value.trim(); const country = countryEl ? countryEl.value : "all"; const url = urlEl.value.trim(); if (!name || !url) { showStatus("Заполни название и ссылку."); return;} let parsed; try { parsed = new URL(url); } catch (e) { showStatus("Неверный формат URL ссылки!"); return; } // Добавляем пользовательский канал customChannels.push({ name: name, country: country, description: "Добавленный пользователем телеканал.", url: url }); nameEl.value = ""; urlEl.value = ""; saveData(); renderChannels(); showStatus("Телеканал успешно добавлен!"); } function deleteChannel(url) { customChannels = customChannels.filter(c => c.url !== url); saveData(); renderChannels(); showStatus("Канал удален."); } // --- ВОСПРОИЗВЕДЕНИЕ И УПРАВЛЕНИЕ HLS ПЛЕЕРОМ --- function playStream(url) { // 1. Очистка предыдущего HLS экземпляра и сброс элемента video if (hls) { hls.destroy(); hls = null; } if (player) { player.pause(); player.removeAttribute("src"); player.load(); } status("Загружаем видеопоток..."); // 2. Инициализация HLS.js (для Chrome/Firefox/Edge) if (window.Hls && Hls.isSupported()) { hls = new Hls({ enableWorker: true, lowLatencyMode: true }); hls.loadSource(url); hls.attachMedia(video); hls.on(Hls.Events.MANIFEST_PARSED, () => { status("Видео загружено! Нажмите ▶."); video.play().catch(e => console.log("Автозапуск заблокирован браузером:", e)); }); hls.on(Hls.Events.ERROR, (event, data) => { if (data.fatal) { switch (data.type) { case Hls.ErrorTypes.NETWORK_ERROR: status("Ошибка сети. Проверь ссылку или доступ к потоку."); hls.startLoad(); break; case Hls.ErrorTypes.MEDIA_ERROR: status("Ошибка видеопотока. Восстанавливаем..."); hls.recoverMediaError(); break; default: status("Ошибка загрузки. Проверь ссылку или доступ к потоку."); hls.destroy(); break; } } }); } // 3. Поддержка нативного HLS (Safari на iOS / macOS) else if (video && video.canPlayType("application/vnd.apple.mpegurl")) { video.src = url; video.addEventListener("loadedmetadata", () => { status("Поток загружен. Нажмите ▶ на видео."); video.play().catch(e => console.log(e)); }, { once: true }); video.addEventListener("error", () => { status("Браузер не смог загрузить этот поток."); }, { once: true }); } else { status("Браузер не поддерживает HLS."); } } // Первичная отрисовка при загрузке страницы document.addEventListener("DOMContentLoaded", () => { renderChannels(); });
